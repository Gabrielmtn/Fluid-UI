const slider = ComponentRegistry.create('slider', { id: 'demoSlider', label: 'Demo Slider', min: 0, max: 100, step: 1, value: 50, unit: '%' }); slider.mount(container); slider.on('change', (value) => { console.log('Value changed:', value); });

2. Synced Slider
